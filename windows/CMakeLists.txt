# 會意注音 (Huiyi, based on AquaType) for Windows: the decoder on the CPU backend (decoder/native/cpu) and the engine (windows/engine).
#   cmake -S windows -B windows/build -G Ninja -DCMAKE_BUILD_TYPE=Release
#   cmake --build windows/build
cmake_minimum_required(VERSION 3.20)
project(huiyi_windows CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(json)

# AQ_TIP_ONLY: just the text service (the 32-bit copy that 32-bit apps load; it talks to the one 64-bit engine)
option(AQ_TIP_ONLY "build only the text service" OFF)
if(NOT AQ_TIP_ONLY)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../decoder/native/cpu decoder)

add_executable(HuiyiEngine engine/engine.cpp engine/aqds.cpp engine/memory.cpp engine/util.cpp tip/settings.cpp)
target_include_directories(HuiyiEngine PRIVATE tip)
target_link_libraries(HuiyiEngine PRIVATE aquadecoder nlohmann_json::nlohmann_json bcrypt advapi32)
target_compile_definitions(HuiyiEngine PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(HuiyiEngine PRIVATE /O2 /utf-8 /EHsc /W3)
endif()
# No console window: the text service and the logon entry start it in the background.  Pipes still work for stdio mode (tests).
target_link_options(HuiyiEngine PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
# the dispatcher and both builds it picks from, next to the engine: a target of its own, so a rebuilt decoder is copied
# even when the engine itself did not relink
add_custom_target(aq_decoder_copy ALL COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:aquadecoder>
  $<TARGET_FILE:aquadecoder_avx2> $<TARGET_FILE:aquadecoder_sse> $<TARGET_FILE_DIR:HuiyiEngine> DEPENDS aquadecoder)
endif()

# the text service's core, without TSF, and its test against the real engine
add_library(aqtip_core STATIC tip/composer.cpp tip/engine_link.cpp tip/settings.cpp)
target_include_directories(aqtip_core PUBLIC tip)
target_link_libraries(aqtip_core PUBLIC nlohmann_json::nlohmann_json ole32 shell32 advapi32)
target_compile_definitions(aqtip_core PRIVATE _CRT_SECURE_NO_WARNINGS)
if(NOT AQ_TIP_ONLY)
add_executable(composer_test tip/composer_test.cpp)
target_link_libraries(composer_test PRIVATE aqtip_core)
endif()
if(MSVC)
  target_compile_options(aqtip_core PRIVATE /O2 /utf-8 /EHsc /W3)
  if(NOT AQ_TIP_ONLY)
    target_compile_options(composer_test PRIVATE /O2 /utf-8 /EHsc)
  endif()
endif()
# the text service (TSF): HuiyiTSF.dll, loaded into every app; registered with regsvr32 (as administrator)
add_library(HuiyiTSF SHARED tip/tsf/dllmain.cpp tip/tsf/tip.cpp tip/tsf/candwin.cpp tip/tsf/langbar.cpp tip/tsf/HuiyiTSF.def tip/tsf/HuiyiTSF.rc)
target_link_libraries(HuiyiTSF PRIVATE aqtip_core dwmapi uuid d2d1 dwrite)
if(MSVC)
  target_compile_options(HuiyiTSF PRIVATE /O2 /utf-8 /EHsc /W3)
  set_property(TARGET HuiyiTSF PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")    # no VC runtime DLL to find inside other apps
  set_property(TARGET aqtip_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  if(NOT AQ_TIP_ONLY)
    set_property(TARGET composer_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  endif()
endif()
# HuiyiSettings.exe: the settings window (right click on the taskbar's 中/英, or Windows Settings' 「選項」)
if(NOT AQ_TIP_ONLY)
add_executable(HuiyiSettings WIN32 settings/settings_app.cpp settings/settings_app.rc settings/settings_app.manifest tip/settings.cpp)
target_include_directories(HuiyiSettings PRIVATE tip)
target_link_libraries(HuiyiSettings PRIVATE comctl32 advapi32 shell32 user32 gdi32)
if(MSVC)
  target_compile_options(HuiyiSettings PRIVATE /O2 /utf-8 /EHsc /W3)
  set_property(TARGET HuiyiSettings PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
endif()
# the candidate window drawn into PNGs (windows/tip/tsf/candwin_preview.cpp)
add_executable(candwin_preview tip/tsf/candwin_preview.cpp tip/tsf/candwin.cpp)
target_link_libraries(candwin_preview PRIVATE aqtip_core dwmapi d2d1 dwrite windowscodecs ole32)
if(MSVC)
  target_compile_options(candwin_preview PRIVATE /O2 /utf-8 /EHsc)
  set_property(TARGET candwin_preview PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
# the README demo: the real composer and engine typing a script, the candidate window drawn into PNGs (windows/tip/tsf/demo_render.cpp)
add_executable(demo_render tip/tsf/demo_render.cpp tip/tsf/candwin.cpp)
target_link_libraries(demo_render PRIVATE aqtip_core dwmapi d2d1 dwrite windowscodecs ole32)
target_compile_options(demo_render PRIVATE /O2 /utf-8 /EHsc)
set_property(TARGET demo_render PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
